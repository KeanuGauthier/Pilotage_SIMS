(function () {
  var LM = window.AJS && AJS.LayerManager && AJS.LayerManager.global;
  if (!LM || typeof LM.popUntil !== 'function') { console.warn('[fix] LayerManager introuvable'); return; }
  if (LM.__fix) { console.info('[fix] deja actif'); return; }
  LM.__fix = true;
  var SURVOL = /^(mouse|pointer)(over|out|enter|leave|move)$/;
  var orig = LM.popUntil;
  LM.popUntil = function (calque) {
    var ev = window.event;
    var ouvert = document.querySelector('section.aui-dialog2[open]');
    var el = calque && calque.jquery ? calque[0] : calque;
    var bloquer = !!(ouvert && ev && SURVOL.test(ev.type));
    console.info('[fix] popUntil', ev ? ev.type : 'aucun evt',
      'calque=' + (el && el.id ? el.id : '?') + (el && el.hasAttribute && el.hasAttribute('open') ? ' (ouvert)' : ' (ferme)'),
      bloquer ? 'BLOQUE' : 'ok');
    if (bloquer) return;
    return orig.apply(this, arguments);
  };
  console.info('[fix] correctif actif');
})();
