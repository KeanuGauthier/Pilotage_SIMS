"""Requirements data of the R4J plugin ("R4J - Requirements Management for Jira").

With R4J, a requirement is a normal Jira issue (archived like any other issue).
What the plugin adds, and keeps in its own tables, is read here through its own
REST API (``/rest/com.easesolutions.jira.plugins.requirements/1.0/...``),
read-only:

* **The requirements tree** of the project: its folders, the issues placed in
  each folder (in their order), and the child requirements of an issue.
* **The baselines** of the project: the frozen copies ("snapshots") of the
  requirements taken at a given date, with the field values they had then.

Requests, once per project (all GET, throttled like every other request):

* 1 request for the tree,
* 1 request for the list of baselines, then 1 request per baseline.

When R4J is not installed (HTTP 404), or the project is not managed with R4J,
nothing more is asked and nothing changes in the archive.

Not covered: coverage and traceability views (they are computed from the issue
links, which are archived with every issue), and the R4J configuration.
"""

from __future__ import annotations

from typing import Any
from urllib.parse import quote

from .jira_client import JiraClient, JiraError
from .logging_setup import get_logger

log = get_logger("r4j")

R4J = "/rest/com.easesolutions.jira.plugins.requirements/1.0"
MAX_DEPTH = 60  # safety net against a tree that would loop on itself

NOT_INSTALLED: dict[str, Any] = {"installed": False}


def _as_list(value: Any) -> list[Any]:
    """R4J sometimes gives one object where a list is expected (``{"childReq": {...}}``)."""
    if isinstance(value, list):
        return value
    return [value] if isinstance(value, dict) else []


def _position(item: dict[str, Any]) -> float:
    position = item.get("position")
    return position if isinstance(position, (int, float)) and not isinstance(position, bool) else 0


def _text(value: Any) -> str:
    return value if isinstance(value, str) else ("" if value is None else str(value))


def _issue(raw: dict[str, Any], depth: int) -> dict[str, Any]:
    children: list[Any] = []
    if depth < MAX_DEPTH:
        nested = raw.get("childReqs")
        if isinstance(nested, dict):
            nested = nested.get("childReq")
        children = _as_list(nested) + _as_list(raw.get("issues"))
    issue_id = raw.get("issueId", raw.get("id"))
    return {
        "key": _text(raw.get("key")),
        "id": None if issue_id is None else str(issue_id),
        "summary": _text(raw.get("summary")),
        "type": _text(raw.get("issueType")),
        "children": [_issue(c, depth + 1) for c in sorted((c for c in children if isinstance(c, dict)), key=_position)
                     if c.get("key")],
    }


def _folder(raw: dict[str, Any], depth: int) -> dict[str, Any]:
    folders = _as_list(raw.get("folders")) if depth < MAX_DEPTH else []
    issues = _as_list(raw.get("issues"))
    folder_id = raw.get("id")
    return {
        "id": None if folder_id is None else str(folder_id),
        "name": _text(raw.get("name_display") or raw.get("name")),
        "description": _text(raw.get("description")),
        "folders": [_folder(f, depth + 1) for f in sorted((f for f in folders if isinstance(f, dict)), key=_position)],
        "issues": [_issue(i, 0) for i in sorted((i for i in issues if isinstance(i, dict)), key=_position)
                   if i.get("key")],
    }


def normalize_tree(raw: Any) -> dict[str, Any] | None:
    """The tree given by R4J, reduced to what is useful in an archive. ``None`` when it is empty."""
    if not isinstance(raw, dict):
        return None
    root = _folder(raw, 0)
    if not root["folders"] and not root["issues"]:
        return None
    return {"folders": root["folders"], "issues": root["issues"]}


def count_tree(tree: dict[str, Any] | None) -> tuple[int, int]:
    """(number of folders, number of issues placed in the tree, child requirements included)."""
    if not tree:
        return 0, 0

    def issues(items: list[dict[str, Any]]) -> int:
        return sum(1 + issues(i.get("children") or []) for i in items)

    folders = issue_count = 0
    stack = [tree]
    while stack:
        node = stack.pop()
        issue_count += issues(node.get("issues") or [])
        folders += len(node.get("folders") or [])
        stack.extend(node.get("folders") or [])
    return folders, issue_count


def export_r4j(client: JiraClient, project_key: str, warnings: list[str]) -> dict[str, Any]:
    """Tree and baselines of one project: ``{"installed", "tree", "baselines"}``."""
    try:
        raw_tree = client.get_json(f"{R4J}/tree/{quote(project_key)}")
    except JiraError as exc:
        if exc.status_code == 404:  # the plugin is not installed: nothing to say
            return dict(NOT_INSTALLED)
        if exc.status_code == 400:  # R4J is installed, but this project is not managed with it
            log.info("[%s] R4J is installed but has no requirements tree for this project.", project_key)
            return {"installed": True, "tree": None, "baselines": []}
        warnings.append(f"R4J requirements: the R4J API did not answer ({exc}). The requirements tree and the "
                        "baselines are NOT archived.")
        log.warning("R4J API not available: %s", exc)
        return dict(NOT_INSTALLED)
    except ValueError:  # not JSON (e.g. an HTML page): treated as "not installed"
        return dict(NOT_INSTALLED)

    tree = normalize_tree(raw_tree)
    baselines = _baselines(client, project_key, warnings)
    folders, issues = count_tree(tree)
    if tree or baselines:
        log.info("[%s] R4J detected: %d folder(s), %d issue(s) in the requirements tree, %d baseline(s).",
                 project_key, folders, issues, len(baselines))
    return {"installed": True, "tree": tree, "baselines": baselines}


def _baselines(client: JiraClient, project_key: str, warnings: list[str]) -> list[dict[str, Any]]:
    try:
        listed = client.get_json(f"{R4J}/baselines/project/{quote(project_key)}")
    except JiraError as exc:
        if exc.status_code not in (400, 404):
            warnings.append(f"R4J baselines: the list could not be read ({exc}). Baselines are NOT archived.")
            log.warning("R4J baselines not available: %s", exc)
        return []
    except ValueError:
        return []
    if isinstance(listed, dict):  # tolerated: {"baselines": [...]}
        listed = listed.get("baselines") or listed.get("values") or []

    result = []
    for baseline in listed if isinstance(listed, list) else []:
        if not isinstance(baseline, dict) or baseline.get("id") is None:
            continue
        item = dict(baseline)
        item["issues"] = []
        try:
            detail = client.get_json(f"{R4J}/baseline/{quote(str(baseline['id']))}")
            if isinstance(detail, dict):
                item.update(detail)
                item["issues"] = [i for i in _as_list(detail.get("issues")) if isinstance(i, dict)]
        except (JiraError, ValueError) as exc:
            warnings.append(f"R4J baseline \"{baseline.get('name')}\" (id {baseline['id']}): its content could not "
                            f"be read ({exc}). Only its name and date are archived.")
            log.warning("R4J baseline %s not readable: %s", baseline["id"], exc)
        result.append(item)
    return result
