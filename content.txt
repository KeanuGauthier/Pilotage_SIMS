"""Logging configuration: readable console output plus a detailed log file."""

from __future__ import annotations

import logging
import sys
from pathlib import Path

from .utils import utc_now

LOGGER_NAME = "jira_archiver"


class _RedactingFilter(logging.Filter):
    """Defense in depth: replaces any occurrence of the secret token in log records."""

    def __init__(self, secret: str) -> None:
        super().__init__()
        self._secret = secret

    def filter(self, record: logging.LogRecord) -> bool:
        if self._secret and len(self._secret) >= 8:
            message = record.getMessage()
            if self._secret in message:
                record.msg = message.replace(self._secret, "***REDACTED***")
                record.args = ()
        return True


def setup_logging(log_dir: Path | None, secret: str = "", verbose: bool = False) -> Path | None:
    """Configure logging. Returns the path of the log file (or ``None``)."""
    logger = logging.getLogger(LOGGER_NAME)
    logger.setLevel(logging.DEBUG)
    logger.handlers.clear()
    redactor = _RedactingFilter(secret)

    console = logging.StreamHandler(sys.stdout)
    console.setLevel(logging.DEBUG if verbose else logging.INFO)
    console.setFormatter(logging.Formatter("%(asctime)s  %(levelname)-7s %(message)s", "%H:%M:%S"))
    console.addFilter(redactor)
    logger.addHandler(console)

    log_file = None
    if log_dir is not None:
        log_dir.mkdir(parents=True, exist_ok=True)
        log_file = log_dir / f"archive_{utc_now().strftime('%Y%m%d_%H%M%S')}.log"
        file_handler = logging.FileHandler(log_file, encoding="utf-8")
        file_handler.setLevel(logging.DEBUG)
        file_handler.setFormatter(
            logging.Formatter("%(asctime)s %(levelname)-7s [%(name)s] %(message)s")
        )
        file_handler.addFilter(redactor)
        logger.addHandler(file_handler)
    return log_file


def get_logger(name: str = "") -> logging.Logger:
    return logging.getLogger(f"{LOGGER_NAME}.{name}" if name else LOGGER_NAME)
