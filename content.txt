"""Convert the HTML that Jira renders (from its wiki markup) into ReportLab flowables.

Jira gives us, for every rich-text field and comment, an HTML version
("renderedFields"). This module turns that HTML into PDF building blocks:
paragraphs, headings, lists, tables, code blocks, quotes and images.

It is deliberately tolerant: unknown tags are treated as plain text containers,
and any piece that ReportLab cannot handle falls back to plain text instead of
failing the whole document.
"""

from __future__ import annotations

import re
from collections.abc import Callable
from html.parser import HTMLParser
from pathlib import Path
from xml.sax.saxutils import escape

from reportlab.lib import colors
from reportlab.lib.styles import ParagraphStyle
from reportlab.platypus import Flowable, Paragraph, Spacer, Table, TableStyle, XPreformatted
from reportlab.platypus.flowables import HRFlowable

from .pdf_styles import BORDER, HEADER_BG, LINK, PdfStyles, make_image

#: Given an ``<img src=...>`` value, returns the image (bytes or local file) or ``None``.
ImageResolver = Callable[[str], bytes | Path | None]

_VOID = {"br", "img", "hr", "input", "meta", "link", "col", "wbr", "area", "base", "source", "param"}
_SKIP = {"script", "style", "head", "title", "noscript", "button", "form", "select",
         "textarea", "svg", "iframe", "object", "embed", "video", "audio", "canvas"}
_AUTO_CLOSE = {"p": {"p"}, "li": {"li"}, "tr": {"tr", "td", "th"}, "td": {"td", "th"},
               "th": {"td", "th"}, "dt": {"dt", "dd"}, "dd": {"dt", "dd"}}
_SIMPLE_INLINE = {
    "b": ("<b>", "</b>"), "strong": ("<b>", "</b>"),
    "i": ("<i>", "</i>"), "em": ("<i>", "</i>"), "cite": ("<i>", "</i>"), "dfn": ("<i>", "</i>"),
    "var": ("<i>", "</i>"),
    "u": ("<u>", "</u>"), "ins": ("<u>", "</u>"),
    "s": ("<strike>", "</strike>"), "del": ("<strike>", "</strike>"), "strike": ("<strike>", "</strike>"),
    "sup": ("<super>", "</super>"), "sub": ("<sub>", "</sub>"),
}
_CODE_INLINE = {"code", "tt", "kbd", "samp"}
_CONTAINERS = {"p", "div", "section", "article", "header", "footer", "main", "center",
               "figure", "figcaption", "dl", "dt", "dd", "body", "html", "nav", "aside", "details", "summary"}
_HEADINGS = {"h1": 1, "h2": 2, "h3": 3, "h4": 4, "h5": 5, "h6": 6}
_WHITESPACE = re.compile(r"\s+")
_TAG = re.compile(r"<[^>]+>")
_COLOR_STYLE = re.compile(r"color\s*:\s*([#a-zA-Z0-9]+)")
_HEX_COLOR = re.compile(r"^#(?:[0-9a-fA-F]{3}|[0-9a-fA-F]{6})$")
_NAMED_COLORS = {name.lower() for name in colors.getAllNamedColors()}


# ------------------------------------------------------------------ HTML tree
class Node:
    __slots__ = ("tag", "attrs", "children")

    def __init__(self, tag: str, attrs: dict[str, str]) -> None:
        self.tag = tag
        self.attrs = attrs
        self.children: list[Node | str] = []


class _TreeBuilder(HTMLParser):
    """Builds a forgiving DOM-like tree (Jira's HTML is not always well formed)."""

    def __init__(self) -> None:
        super().__init__(convert_charrefs=True)
        self.root = Node("root", {})
        self._stack = [self.root]

    def handle_starttag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
        closes = _AUTO_CLOSE.get(tag)
        if closes and self._stack[-1].tag in closes:
            self._stack.pop()
        node = Node(tag, {key: value or "" for key, value in attrs})
        self._stack[-1].children.append(node)
        if tag not in _VOID:
            self._stack.append(node)

    def handle_startendtag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
        self.handle_starttag(tag, attrs)
        if tag not in _VOID and self._stack[-1].tag == tag:
            self._stack.pop()

    def handle_endtag(self, tag: str) -> None:
        if any(node.tag == tag for node in self._stack[1:]):
            while self._stack[-1].tag != tag:
                self._stack.pop()
            self._stack.pop()

    def handle_data(self, data: str) -> None:
        self._stack[-1].children.append(data)


def parse_html(html: str) -> Node:
    builder = _TreeBuilder()
    builder.feed(html or "")
    builder.close()
    return builder.root


def text_content(node: Node | str) -> str:
    """Raw text of a node (``<br>`` becomes a new line)."""
    if isinstance(node, str):
        return node
    if node.tag == "br":
        return "\n"
    if node.tag in _SKIP:
        return ""
    return "".join(text_content(child) for child in node.children)


def strip_tags(markup: str) -> str:
    return _TAG.sub("", markup)


# ------------------------------------------------------------- the converter
class _Context:
    """State of one container being rendered (a cell, a list item, the root...)."""

    def __init__(self, style: ParagraphStyle, width: float) -> None:
        self.style = style
        self.width = width
        self.out: list[Flowable] = []
        self.buffer: list[str] = []          # inline markup waiting to become a Paragraph
        self.open_tags: list[tuple[str, str]] = []  # inline tags currently open


class HtmlToFlowables:
    """Converts Jira HTML into a list of ReportLab flowables.

    :param styles: shared PDF styles.
    :param resolve_image: callback returning image data for an ``<img src>``.
    :param safe_mode: simpler output (tables flattened to text). Used as a
        fallback when a document cannot be laid out normally.
    """

    def __init__(self, styles: PdfStyles, resolve_image: ImageResolver | None = None,
                 safe_mode: bool = False) -> None:
        self.styles = styles
        self.resolve_image = resolve_image
        self.safe_mode = safe_mode
        self._pending_bullet: str | None = None

    # -------------------------------------------------------------- public
    def convert(self, html: str, width: float, style: ParagraphStyle | None = None) -> list[Flowable]:
        root = parse_html(html)
        return self._render(root.children, style or self.styles.body, width)

    def paragraph(self, markup: str, style: ParagraphStyle) -> Paragraph:
        """Create a Paragraph; if the markup is rejected, fall back to escaped plain text."""
        bullet = self._pending_bullet
        self._pending_bullet = None
        try:
            return Paragraph(markup, style, bulletText=bullet)
        except Exception:
            return Paragraph(escape(strip_tags(markup)), style, bulletText=bullet)

    # ------------------------------------------------------------ internals
    def _render(self, nodes: list[Node | str], style: ParagraphStyle, width: float) -> list[Flowable]:
        ctx = _Context(style, width)
        for node in nodes:
            self._visit(node, ctx)
        self._flush(ctx)
        return ctx.out

    def _flush(self, ctx: _Context) -> None:
        closing = "".join(close for _, close in reversed(ctx.open_tags))
        markup = "".join(ctx.buffer) + closing
        ctx.buffer = [opening for opening, _ in ctx.open_tags]
        markup = re.sub(r"^(\s|<br/>)+", "", markup)
        markup = re.sub(r"(\s|<br/>)+$", "", markup)
        if strip_tags(markup).strip():
            ctx.out.append(self.paragraph(markup, ctx.style))

    def _visit(self, node: Node | str, ctx: _Context) -> None:
        if isinstance(node, str):
            text = _WHITESPACE.sub(" ", node)
            if text:
                ctx.buffer.append(escape(text))
            return

        tag = node.tag
        if tag in _SKIP:
            return
        if tag == "br":
            ctx.buffer.append("<br/>")
            return
        if tag == "img":
            self._visit_image(node, ctx)
            return
        if tag in _HEADINGS:
            self._flush(ctx)
            ctx.out.extend(self._render(node.children, self.styles.heading(_HEADINGS[tag]), ctx.width))
            return
        if tag in ("ul", "ol"):
            self._flush(ctx)
            ctx.out.extend(self._list(node, ctx))
            return
        if tag == "table":
            self._flush(ctx)
            ctx.out.extend(self._table(node, ctx))
            return
        if tag == "pre":
            self._flush(ctx)
            ctx.out.extend(self._preformatted(node, ctx))
            return
        if tag == "blockquote":
            self._flush(ctx)
            quote_style = self.styles.indented(ctx.style, 12, quote=True)
            ctx.out.extend(self._render(node.children, quote_style, ctx.width - 12))
            return
        if tag == "hr":
            self._flush(ctx)
            ctx.out.append(HRFlowable(width="100%", thickness=0.5, color=BORDER, spaceBefore=3, spaceAfter=3))
            return
        if tag in _CONTAINERS or tag == "li":
            self._flush(ctx)
            css_class = node.attrs.get("class", "")
            style = self.styles.panel_header if "panelHeader" in css_class else ctx.style
            if tag == "dd":
                style = self.styles.indented(ctx.style, 12)
            ctx.out.extend(self._render(node.children, style, ctx.width))
            return

        opening, closing = self._inline_tags(node)
        if opening:
            ctx.buffer.append(opening)
            ctx.open_tags.append((opening, closing))
        for child in node.children:
            self._visit(child, ctx)
        if opening:
            ctx.open_tags.pop()
            ctx.buffer.append(closing)

    def _inline_tags(self, node: Node) -> tuple[str, str]:
        tag = node.tag
        if tag in _SIMPLE_INLINE:
            return _SIMPLE_INLINE[tag]
        if tag in _CODE_INLINE:
            return f'<font face="{self.styles.mono}">', "</font>"
        if tag == "a":
            href = node.attrs.get("href", "").strip()
            if href.lower().startswith(("http://", "https://", "mailto:")):
                safe_href = escape(href, {'"': "&quot;"})
                return f'<a href="{safe_href}" color="{LINK.hexval()}">', "</a>"
            return "", ""
        if tag in ("span", "font"):
            color = node.attrs.get("color", "")
            match = _COLOR_STYLE.search(node.attrs.get("style", ""))
            if match:
                color = match.group(1)
            color = color.strip().lower()
            if _HEX_COLOR.match(color) or color in _NAMED_COLORS:
                return f'<font color="{color}">', "</font>"
        return "", ""

    def _visit_image(self, node: Node, ctx: _Context) -> None:
        src = node.attrs.get("src", "")
        css_class = node.attrs.get("class", "")
        if "emoticon" in css_class or "/images/icons/" in src:
            # Small Jira icons (smileys, status icons): rendered as text, e.g. "(smile)".
            name = node.attrs.get("alt") or Path(src.split("?")[0]).stem
            if name:
                ctx.buffer.append(escape(f"({name.strip('()')})"))
            return
        self._flush(ctx)
        name = node.attrs.get("alt") or Path(src.split("?")[0]).name or "image"
        data = self.resolve_image(src) if self.resolve_image else None
        image = make_image(data, ctx.width) if data is not None else None
        if image is None:
            ctx.out.append(self.paragraph(f"<i>[Image not available: {escape(name)}]</i>", self.styles.muted))
        else:
            ctx.out.append(image)
            ctx.out.append(Spacer(1, 3))

    def _list(self, node: Node, ctx: _Context) -> list[Flowable]:
        ordered = node.tag == "ol"
        try:
            number = int(node.attrs.get("start", "1"))
        except ValueError:
            number = 1
        indent = 14
        style = self.styles.indented(ctx.style, indent)
        nested = ctx.style.leftIndent > 0
        out: list[Flowable] = []
        for child in node.children:
            if isinstance(child, str):
                if child.strip():
                    out.extend(self._render([child], style, ctx.width - indent))
                continue
            if child.tag != "li":
                out.extend(self._render([child], style, ctx.width - indent))
                continue
            self._pending_bullet = f"{number}." if ordered else ("–" if nested else "•")
            number += 1
            out.extend(self._render(child.children, style, ctx.width - indent))
            self._pending_bullet = None
        return out

    def _preformatted(self, node: Node, ctx: _Context) -> list[Flowable]:
        text = text_content(node).expandtabs(4).strip("\n")
        if not text.strip():
            return []
        style = self.styles.code
        max_chars = max(20, int((ctx.width - 10) / (style.fontSize * 0.6)))
        lines: list[str] = []
        for line in text.splitlines():
            while len(line) > max_chars:  # hard-wrap very long lines, they would overflow the page
                lines.append(line[:max_chars])
                line = line[max_chars:]
            lines.append(line)
        markup = escape("\n".join(lines))
        try:
            return [XPreformatted(markup, style)]
        except Exception:
            return [self.paragraph(markup.replace("\n", "<br/>"), style)]

    def _table(self, node: Node, ctx: _Context) -> list[Flowable]:
        rows: list[list[Node]] = []

        def collect(parent: Node) -> None:
            for child in parent.children:
                if isinstance(child, str):
                    continue
                if child.tag == "tr":
                    cells = [c for c in child.children if isinstance(c, Node) and c.tag in ("td", "th")]
                    if cells:
                        rows.append(cells)
                elif child.tag in ("thead", "tbody", "tfoot"):
                    collect(child)

        collect(node)
        if not rows:
            return []

        if self.safe_mode:  # flattened: one paragraph per row, cells separated by "|"
            out = []
            for cells in rows:
                texts = [escape(_WHITESPACE.sub(" ", text_content(c)).strip()) for c in cells]
                out.append(self.paragraph(" | ".join(texts), self.styles.small))
            return out

        column_count = max(len(cells) for cells in rows)
        column_width = ctx.width / column_count
        padding = 3
        data: list[list[object]] = []
        commands: list[tuple] = [
            ("GRID", (0, 0), (-1, -1), 0.4, BORDER),
            ("VALIGN", (0, 0), (-1, -1), "TOP"),
            ("LEFTPADDING", (0, 0), (-1, -1), padding),
            ("RIGHTPADDING", (0, 0), (-1, -1), padding),
            ("TOPPADDING", (0, 0), (-1, -1), 2),
            ("BOTTOMPADDING", (0, 0), (-1, -1), 2),
        ]
        for row_index, cells in enumerate(rows):
            row: list[object] = []
            for col_index, cell in enumerate(cells):
                header = cell.tag == "th"
                style = self.styles.cell_header if header else self.styles.cell
                content = self._render(cell.children, style, column_width - 2 * padding)
                row.append(content or "")
                if header:
                    commands.append(("BACKGROUND", (col_index, row_index), (col_index, row_index), HEADER_BG))
            row.extend([""] * (column_count - len(row)))
            data.append(row)

        header_rows = 1 if all(c.tag == "th" for c in rows[0]) and len(rows) > 1 else 0
        table = Table(data, colWidths=[column_width] * column_count, repeatRows=header_rows, hAlign="LEFT")
        table.setStyle(TableStyle(commands))
        return [table, Spacer(1, 5)]
