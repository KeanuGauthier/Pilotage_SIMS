"""Fonts, colors and paragraph styles shared by the PDF renderer.

ReportLab's built-in fonts (Helvetica/Courier) only cover Western European
characters. To render accents, umlauts and most other characters correctly, a
TrueType font installed on the machine is used when available (Arial and
Consolas on Windows), with a silent fallback to the built-in fonts.
"""

from __future__ import annotations

import io
import os
from dataclasses import dataclass, field
from functools import cache
from pathlib import Path

from reportlab.lib import colors
from reportlab.lib.styles import ParagraphStyle
from reportlab.lib.units import mm
from reportlab.pdfbase import pdfmetrics
from reportlab.pdfbase.ttfonts import TTFont
from reportlab.platypus import Flowable, Image

from .logging_setup import get_logger

log = get_logger("pdf")

# --------------------------------------------------------------------- colors
PRIMARY = colors.HexColor("#0B3D91")
TEXT = colors.HexColor("#172B4D")
MUTED = colors.HexColor("#5E6C84")
BORDER = colors.HexColor("#C1C7D0")
LIGHT_BG = colors.HexColor("#F4F5F7")
HEADER_BG = colors.HexColor("#E9EBEE")
LINK = colors.HexColor("#0052CC")

# --------------------------------------------------------------------- fonts
_FONT_CANDIDATES = {
    # family: (regular, bold, italic, bold-italic) file names, searched in font folders
    "sans": [
        ("arial.ttf", "arialbd.ttf", "ariali.ttf", "arialbi.ttf"),            # Windows
        ("Arial.ttf", "Arial Bold.ttf", "Arial Italic.ttf", "Arial Bold Italic.ttf"),  # macOS
        ("DejaVuSans.ttf", "DejaVuSans-Bold.ttf", "DejaVuSans-Oblique.ttf", "DejaVuSans-BoldOblique.ttf"),
    ],
    "mono": [
        ("consola.ttf", "consolab.ttf", "consolai.ttf", "consolaz.ttf"),       # Windows
        ("Courier New.ttf", "Courier New Bold.ttf", "Courier New Italic.ttf", "Courier New Bold Italic.ttf"),
        ("DejaVuSansMono.ttf", "DejaVuSansMono-Bold.ttf",
         "DejaVuSansMono-Oblique.ttf", "DejaVuSansMono-BoldOblique.ttf"),
    ],
}
_BUILTIN = {
    "sans": ("Helvetica", "Helvetica-Bold", "Helvetica-Oblique", "Helvetica-BoldOblique"),
    "mono": ("Courier", "Courier-Bold", "Courier-Oblique", "Courier-BoldOblique"),
}


def _font_dirs() -> list[Path]:
    dirs = [Path(os.environ.get("WINDIR", r"C:\Windows")) / "Fonts"]
    local = os.environ.get("LOCALAPPDATA")
    if local:
        dirs.append(Path(local) / "Microsoft" / "Windows" / "Fonts")
    dirs += [
        Path("/System/Library/Fonts/Supplemental"),
        Path("/Library/Fonts"),
        Path("/usr/share/fonts/truetype/dejavu"),
        Path("/usr/share/fonts/dejavu"),
    ]
    return [d for d in dirs if d.is_dir()]


def _find(file_name: str) -> Path | None:
    for directory in _font_dirs():
        candidate = directory / file_name
        if candidate.is_file():
            return candidate
    return None


@cache
def register_fonts() -> dict[str, str]:
    """Register the best available fonts. Returns ``{"sans": name, "mono": name}``."""
    result: dict[str, str] = {}
    for family, candidates in _FONT_CANDIDATES.items():
        registered = False
        for files in candidates:
            paths = [_find(name) for name in files]
            if paths[0] is None:
                continue
            paths = [p or paths[0] for p in paths]  # missing variants fall back to regular
            base = f"Archive{family.capitalize()}"
            names = (base, f"{base}-Bold", f"{base}-Italic", f"{base}-BoldItalic")
            try:
                for name, path in zip(names, paths, strict=True):
                    pdfmetrics.registerFont(TTFont(name, str(path)))
            except Exception as exc:  # corrupted / unsupported font file: try next candidate
                log.debug("Font %s unusable: %r", paths[0], exc)
                continue
            pdfmetrics.registerFontFamily(base, normal=names[0], bold=names[1], italic=names[2], boldItalic=names[3])
            result[family] = base
            registered = True
            break
        if not registered:
            regular, bold, italic, bold_italic = _BUILTIN[family]
            pdfmetrics.registerFontFamily(regular, normal=regular, bold=bold, italic=italic, boldItalic=bold_italic)
            result[family] = regular
    return result


def bold_font(family_name: str) -> str:
    return {"Helvetica": "Helvetica-Bold", "Courier": "Courier-Bold"}.get(family_name, f"{family_name}-Bold")


# --------------------------------------------------------------------- styles
@dataclass
class PdfStyles:
    sans: str
    mono: str
    body: ParagraphStyle = field(init=False)
    small: ParagraphStyle = field(init=False)
    muted: ParagraphStyle = field(init=False)
    title: ParagraphStyle = field(init=False)
    issue_key: ParagraphStyle = field(init=False)
    section: ParagraphStyle = field(init=False)
    label: ParagraphStyle = field(init=False)
    cell: ParagraphStyle = field(init=False)
    cell_header: ParagraphStyle = field(init=False)
    code: ParagraphStyle = field(init=False)
    comment_header: ParagraphStyle = field(init=False)
    panel_header: ParagraphStyle = field(init=False)
    _derived: dict = field(init=False, default_factory=dict)

    def __post_init__(self) -> None:
        bold = bold_font(self.sans)
        self.body = ParagraphStyle("body", fontName=self.sans, fontSize=9, leading=12.5,
                                   textColor=TEXT, spaceAfter=4, splitLongWords=1)
        self.small = ParagraphStyle("small", parent=self.body, fontSize=8, leading=10.5, spaceAfter=2)
        self.muted = ParagraphStyle("muted", parent=self.small, textColor=MUTED)
        self.title = ParagraphStyle("title", parent=self.body, fontName=bold, fontSize=16,
                                    leading=20, spaceAfter=4, textColor=TEXT)
        self.issue_key = ParagraphStyle("issue_key", parent=self.body, fontName=bold, fontSize=10,
                                        textColor=PRIMARY, spaceAfter=2)
        self.section = ParagraphStyle("section", parent=self.body, fontName=bold, fontSize=11.5,
                                      leading=15, textColor=PRIMARY, spaceBefore=10, spaceAfter=5)
        self.label = ParagraphStyle("label", parent=self.small, fontName=bold, textColor=MUTED, spaceAfter=0)
        self.cell = ParagraphStyle("cell", parent=self.small, spaceAfter=1)
        self.cell_header = ParagraphStyle("cell_header", parent=self.cell, fontName=bold)
        self.code = ParagraphStyle("code", parent=self.body, fontName=self.mono, fontSize=7.5, leading=9.5,
                                   backColor=LIGHT_BG, borderPadding=4, spaceBefore=4, spaceAfter=8)
        self.comment_header = ParagraphStyle("comment_header", parent=self.small, textColor=MUTED,
                                             spaceBefore=6, spaceAfter=3)
        self.panel_header = ParagraphStyle("panel_header", parent=self.body, fontName=bold)

    def heading(self, level: int) -> ParagraphStyle:
        sizes = {1: 13, 2: 12, 3: 11, 4: 10, 5: 9.5, 6: 9}
        key = f"h{level}"
        if key not in self._derived:
            size = sizes.get(level, 9)
            self._derived[key] = ParagraphStyle(key, parent=self.body, fontName=bold_font(self.sans),
                                                fontSize=size, leading=size * 1.3, spaceBefore=5, spaceAfter=3)
        return self._derived[key]

    def indented(self, base: ParagraphStyle, indent: float, quote: bool = False) -> ParagraphStyle:
        """A copy of *base* shifted right by *indent* points (lists, quotes)."""
        key = (base.name, indent, quote)
        if key not in self._derived:
            style = ParagraphStyle(f"{base.name}_i{indent}{'q' if quote else ''}", parent=base,
                                   leftIndent=base.leftIndent + indent,
                                   bulletIndent=base.leftIndent + 2,
                                   bulletFontName=self.sans)
            if quote:
                style.textColor = MUTED
            self._derived[key] = style
        return self._derived[key]


def make_styles() -> PdfStyles:
    fonts = register_fonts()
    return PdfStyles(sans=fonts["sans"], mono=fonts["mono"])


# --------------------------------------------------------------------- images
MAX_IMAGE_PIXELS = 1600  # images are downscaled to this size before embedding (keeps PDFs small)


def make_image(source: bytes | Path, max_width: float, max_height: float = 200 * mm) -> Flowable | None:
    """Build a ReportLab image that fits in the given box, or ``None`` if unreadable."""
    from PIL import Image as PILImage  # imported lazily, Pillow is a ReportLab dependency

    try:
        opened = PILImage.open(io.BytesIO(source) if isinstance(source, bytes) else source)
        with opened as img:
            original_format = img.format
            img.seek(0)  # first frame of animated GIFs
            if img.mode in ("P", "PA", "LA", "RGBA"):
                img = img.convert("RGBA")
            elif img.mode not in ("RGB", "L"):
                img = img.convert("RGB")
            if max(img.size) > MAX_IMAGE_PIXELS:
                img.thumbnail((MAX_IMAGE_PIXELS, MAX_IMAGE_PIXELS))
            buffer = io.BytesIO()
            if original_format == "JPEG" and img.mode != "RGBA":
                img.save(buffer, "JPEG", quality=85)
            else:
                img.save(buffer, "PNG", optimize=True)
            width_px, height_px = img.size
    except Exception:
        return None

    width, height = width_px * 0.75, height_px * 0.75  # 96 dpi screen pixels -> PDF points
    scale = min(1.0, max_width / width, max_height / height)
    buffer.seek(0)
    flowable = Image(buffer, width=width * scale, height=height * scale)
    flowable.hAlign = "LEFT"
    return flowable
