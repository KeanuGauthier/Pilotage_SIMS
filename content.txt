(function () {
  var LM = window.AJS && AJS.LayerManager && AJS.LayerManager.global;
  if (!LM || typeof LM.popUntil !== 'function') {
    console.warn('[fix] LayerManager AUI introuvable, correctif non applique');
    return;
  }
  if (LM.__fixFocus) { console.info('[fix] deja actif'); return; }
  LM.__fixFocus = true;

  var FOCUS = { focusin: 1, focusout: 1, focus: 1, blur: 1 };
  var orig = LM.popUntil;

  LM.popUntil = function () {
    var ev = window.event;
    var dialogOuvert = document.querySelector('section.aui-dialog2[open]');
    if (ev && FOCUS[ev.type] && dialogOuvert) {
      console.info('[fix] fermeture bloquee (' + ev.type + ') sur', dialogOuvert.id);
      return;
    }
    return orig.apply(this, arguments);
  };

  console.info('[fix] correctif actif');
})();
