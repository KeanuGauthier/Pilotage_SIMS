@echo off
REM Archives one or more Jira projects.
REM   Double-click        : asks for the project key
REM   archive.bat ABC     : archives project ABC
REM   archive.bat --file projects.txt --max-issues 1000
REM   archive.bat --help  : shows every option
setlocal
cd /d "%~dp0"
if not exist ".venv\Scripts\python.exe" (
    echo The tool is not installed yet: double-click setup.bat first.
    pause
    exit /b 1
)
if "%~1"=="" (
    set /p "PROJECT=Key of the project to archive (e.g. ABC): "
) else (
    set "PROJECT="
)
if "%~1"=="" (
    ".venv\Scripts\python.exe" archive_project.py "%PROJECT%"
) else (
    ".venv\Scripts\python.exe" archive_project.py %*
)
set "RESULT=%ERRORLEVEL%"
echo.
pause
exit /b %RESULT%
