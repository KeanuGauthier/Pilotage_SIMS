"""Check that the connection to Jira works, step by step, with clear explanations.

Usage:

    py check_connection.py            # checks server, certificate, token, admin rights
    py check_connection.py PROJKEY    # also checks access to one project

Nothing is ever modified in Jira: this script only reads information.
"""

from __future__ import annotations

import platform
import sys
from importlib import metadata
from pathlib import Path
from urllib.parse import quote

from jira_archiver import __version__
from jira_archiver.bootstrap import pause_if_double_clicked, started_by_double_click, use_project_venv

PROJECT_ROOT = Path(__file__).resolve().parent

# Must run before the third-party libraries are imported (see bootstrap.py).
use_project_venv(PROJECT_ROOT)

# Standard library only: these imports work even when the libraries are missing.
from jira_archiver.config import ConfigError, load_settings  # noqa: E402
from jira_archiver.utils import PROJECT_KEY_RE  # noqa: E402

OK, FAIL, WARN = "[ OK ]", "[FAIL]", "[WARN]"


def report(status: str, message: str, hint: str = "") -> None:
    print(f"{status} {message}")
    if hint:
        print(f"       -> {hint}")


def package_version(name: str) -> str | None:
    try:
        return metadata.version(name)
    except metadata.PackageNotFoundError:
        return None


def main(argv: list[str]) -> int:
    print(f"Jira connection check (automatic-jira-archiving {__version__})")
    print("-" * 60)

    # 1. Python and libraries ------------------------------------------------
    report(OK, f"Python {platform.python_version()} on {platform.system()} {platform.release()}")
    missing = []
    for package in ("requests", "reportlab", "pillow", "pip-system-certs"):
        version = package_version(package)
        if version:
            report(OK, f"Library {package} {version}")
        else:
            missing.append(package)
            report(FAIL if package != "pip-system-certs" else WARN, f"Library {package} is not installed",
                   "Run the installation again: py install.py   (see README.md, section 1)")
    if any(p != "pip-system-certs" for p in missing):
        return 1

    # 2. Configuration -------------------------------------------------------
    try:
        settings = load_settings(PROJECT_ROOT)
    except ConfigError as exc:
        report(FAIL, "Configuration", str(exc))
        return 1
    report(OK, f"Configuration loaded: JIRA_BASE_URL = {settings.base_url}")
    report(OK, f"Token present ({len(settings.token)} characters, not displayed for security)")

    from jira_archiver.jira_client import JiraClient, JiraError  # needs "requests", checked above

    client = JiraClient(settings)

    # 3. Server reachable + certificate --------------------------------------
    try:
        info = client.server_info()
    except JiraError as exc:
        hint = ("Check the URL in .env, that you are on the company network (or VPN), "
                "and that 'pip_system_certs' is installed for the internal certificate.")
        report(FAIL, f"Cannot reach the Jira server: {exc}", hint)
        return 1
    security = "HTTPS certificate valid" if settings.base_url.startswith("https://") else "WARNING: not HTTPS"
    report(OK, f"Server reachable ({security}): {info.get('serverTitle', 'Jira')} "
               f"version {info.get('version')} ({info.get('deploymentType', 'unknown type')})")
    if not str(info.get("version", "")).startswith("11."):
        report(WARN, "This tool was written for Jira Data Center 11.x; other versions may differ.")

    # 4. Token valid ------------------------------------------------------------
    try:
        me = client.myself()
    except JiraError as exc:
        report(FAIL, f"Authentication refused: {exc}",
               "Create a new Personal Access Token in Jira (Profile > Personal Access Tokens) "
               "and put it in .env (JIRA_PAT=...).")
        return 1
    report(OK, f"Authenticated as {me.get('displayName')} ({me.get('name')})")

    # 5. Admin rights ---------------------------------------------------------
    try:
        perms = client.get_json(f"{client.API}/mypermissions", {"permissions": "ADMINISTER"})
        is_admin = bool(perms.get("permissions", {}).get("ADMINISTER", {}).get("havePermission"))
    except JiraError:
        is_admin = False
    if is_admin:
        report(OK, "Account has Jira administrator rights")
    else:
        report(WARN, "Account is NOT a Jira administrator",
               "Some project configuration (schemes, roles) may be missing from the archives.")

    # 6. Optional project ------------------------------------------------------
    if argv:
        project_ref = argv[0]
        if not (project_ref.isdigit() or PROJECT_KEY_RE.match(project_ref)):
            report(FAIL, f"'{project_ref}' is not a valid project key or ID")
            return 1
        try:
            project = client.get_json(f"{client.API}/project/{quote(project_ref)}")
            total = client.search(f'project = "{project["key"]}"', 0, 0, ["key"]).get("total", 0)
        except JiraError as exc:
            report(FAIL, f"Cannot read project {project_ref}: {exc}")
            return 1
        report(OK, f"Project {project['key']} - {project.get('name')}: {total} issue(s) visible")

    print("-" * 60)
    print("All checks passed. You can now run: py archive_project.py <PROJECT KEY>")
    return 0


if __name__ == "__main__":
    _double_clicked = started_by_double_click()
    _exit_code = main(sys.argv[1:])
    pause_if_double_clicked(_double_clicked)
    sys.exit(_exit_code)
