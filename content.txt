(function () {
  var opened = {}, retried = {};
  new MutationObserver(function (ms) {
    ms.forEach(function (m) {
      var d = m.target;
      if (!d.matches('section.aui-dialog2')) return;
      if (d.hasAttribute('open')) { opened[d.id] = Date.now(); return; }
      var t = opened[d.id], now = Date.now();
      delete opened[d.id];
      if (!t || now - t > 500) return;
      if (retried[d.id] && now - retried[d.id] < 2000) return;
      retried[d.id] = now;
      setTimeout(function () { AJS.dialog2(d).show(); }, 50);
    });
  }).observe(document.body, { attributes: true, subtree: true, attributeFilter: ['open'] });
})();
