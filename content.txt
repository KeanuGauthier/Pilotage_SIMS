"""Install automatic-jira-archiving (run it once, and again after each ``git pull``).

Usage (in a terminal opened in the project folder):

    py install.py              # recommended: private Python environment in the .venv folder
    py install.py --no-venv    # if the .venv Python is blocked on your PC (see README.md)

What it does:

1. checks the Python version;
2. creates a private Python environment in the ``.venv`` folder (unless ``--no-venv``);
3. installs the libraries listed in ``requirements.txt`` with pip;
4. creates the configuration file ``.env`` from ``.env.example`` (an existing ``.env`` is kept).

This script uses the Python standard library only.
"""

from __future__ import annotations

import argparse
import shutil
import subprocess
import sys
from pathlib import Path

from jira_archiver.bootstrap import pause_if_double_clicked, started_by_double_click, venv_python

PROJECT_ROOT = Path(__file__).resolve().parent
MIN_PYTHON = (3, 10)


def step(message: str) -> None:
    print(f"\n--- {message}")


def run(command: list[str]) -> bool:
    """Run *command* (a fixed list of arguments, no shell). Return True on success."""
    sys.stdout.flush()  # keep our messages in order with the output of the command
    try:
        return subprocess.run(command, cwd=PROJECT_ROOT, check=False).returncode == 0  # noqa: S603
    except OSError as exc:
        print(f"[ERROR] Could not start {command[0]}: {exc}")
        return False


def install(use_venv: bool) -> bool:
    print("=== automatic-jira-archiving installation ===")

    step("Checking Python")
    print(f"Python {sys.version.split()[0]} ({sys.executable})")
    if sys.version_info < MIN_PYTHON:
        print(f"[ERROR] Python {MIN_PYTHON[0]}.{MIN_PYTHON[1]} or newer is required. Install Python 3.12.")
        return False

    if use_venv:
        python = venv_python(PROJECT_ROOT)
        if python.is_file():
            step("The private Python environment (.venv folder) already exists")
        else:
            step("Creating the private Python environment in the .venv folder")
            if not run([sys.executable, "-m", "venv", str(PROJECT_ROOT / ".venv")]):
                print("[ERROR] The .venv folder could not be created.")
                print("        Try: py install.py --no-venv   (see README.md)")
                return False
        if not run([str(python), "-c", "pass"]):
            print("[ERROR] The Python of the .venv folder cannot start (maybe blocked by a company policy).")
            print("        Delete the .venv folder, then run: py install.py --no-venv")
            return False
        pip_command = [str(python), "-m", "pip", "install", "-r", "requirements.txt"]
    else:
        step("Installing without .venv: the libraries go to your user profile")
        if (PROJECT_ROOT / ".venv").exists():
            print("[ERROR] A .venv folder exists: the scripts would keep trying to use it.")
            print("        Delete the .venv folder, then run this command again.")
            return False
        pip_command = [sys.executable, "-m", "pip", "install", "--user", "-r", "requirements.txt"]

    step("Installing the libraries listed in requirements.txt (this can take a minute)")
    if not run(pip_command):
        print("[ERROR] pip could not install the libraries. Read the messages above.")
        print('        See "Troubleshooting" in README.md (internal package mirror).')
        return False

    env_file = PROJECT_ROOT / ".env"
    if env_file.exists():
        step("Configuration file .env already exists: kept as it is")
    else:
        shutil.copyfile(PROJECT_ROOT / ".env.example", env_file)
        step("Configuration file .env created")

    print("\n=== Installation finished ===")
    print("Next steps:")
    print('  1. Open the ".env" file with Notepad and fill in JIRA_BASE_URL and JIRA_PAT.')
    print("  2. Test the connection:  py check_connection.py")
    return True


def main(argv: list[str]) -> int:
    parser = argparse.ArgumentParser(description="Install automatic-jira-archiving and its libraries.")
    parser.add_argument("--no-venv", action="store_true",
                        help="Install the libraries in your user profile instead of a private .venv folder.")
    args = parser.parse_args(argv)
    if install(use_venv=not args.no_venv):
        return 0
    print("\n=== Installation FAILED: read the messages above, and \"Troubleshooting\" in README.md ===")
    return 1


if __name__ == "__main__":
    _double_clicked = started_by_double_click()
    _exit_code = main(sys.argv[1:])
    pause_if_double_clicked(_double_clicked)
    sys.exit(_exit_code)
