var o = AJS.dialog2;
AJS.dialog2 = function (el) {
  var r = o.apply(this, arguments);
  ['show', 'hide'].forEach(function (m) {
    var f = r[m];
    if (typeof f !== 'function') return;
    r[m] = function () { console.trace('dialog2.' + m, el); return f.apply(this, arguments); };
  });
  return r;
};
Object.keys(o).forEach(function (k) { AJS.dialog2[k] = o[k]; });
