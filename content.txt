"""Readable text for the Jira "Development" field (summary of branches, commits, pull requests...).

Jira Data Center returns this field as the text dump of an internal Java object.
:func:`format_dev_summary` turns it into a short sentence such as
``1 branch, 3 commits, 1 pull request (1 merged) - Bitbucket Server``. It is used
for the full-text search of the viewer; ``viewer_assets/app.js`` applies the same
rules for the display (keep both in line).
"""

from __future__ import annotations

import json
import re
from typing import Any

#: Part of the custom field type of the "Development" field (…:devsummarycf).
DEV_SUMMARY_TYPE = "devsummary"
#: Sections of the "Development" summary, in the order they are shown, with their unit.
_DEV_SECTIONS = (
    ("branch", "branch", "branches"),
    ("repository", "commit", "commits"),
    ("pullrequest", "pull request", "pull requests"),
    ("review", "review", "reviews"),
    ("build", "build", "builds"),
    ("deployment-environment", "deployment environment", "deployment environments"),
)
_DEV_SECTION_RE = re.compile(r"\b(" + "|".join(re.escape(s[0]) for s in _DEV_SECTIONS) + r")=com\.atlassian")


def is_dev_summary(value: Any) -> bool:
    """True for the value of the "Development" field (summary of branches, commits, pull requests...).

    Jira Data Center returns this field as the text dump of an internal Java object
    (``{summaryBean=com.atlassian...SummaryBean@1a2b[...], devSummaryJson={...}}``).
    """
    if isinstance(value, str):
        return value.startswith("{summaryBean=") or "devSummaryJson=" in value
    if isinstance(value, dict):
        return "cachedValue" in value or isinstance(value.get("summary"), dict) and "pullrequest" in value["summary"]
    return False


def _dev_summary_data(value: Any) -> dict[str, Any] | None:
    """The ``summary`` part of a "Development" value, from its embedded JSON, or None."""
    data: Any = value
    if isinstance(value, str):
        start = value.find("devSummaryJson=")
        if start < 0:
            return None
        try:
            data, _ = json.JSONDecoder().raw_decode(value, start + len("devSummaryJson="))
        except ValueError:
            return None
    if isinstance(data, dict) and isinstance(data.get("cachedValue"), dict):
        data = data["cachedValue"]
    summary = data.get("summary") if isinstance(data, dict) else None
    return summary if isinstance(summary, dict) else None


def _dev_counts_from_text(text: str) -> dict[str, dict[str, Any]]:
    """Fallback when there is no embedded JSON: read the counters in the Java text dump."""
    found = list(_DEV_SECTION_RE.finditer(text))
    sections: dict[str, dict[str, Any]] = {}
    for n, match in enumerate(found):
        end = found[n + 1].start() if n + 1 < len(found) else len(text)
        part = text[match.end():end]
        count = re.search(r"\bcount=(\d+)", part)
        state = re.search(r"\bstate='?([A-Za-z]+)", part)
        sections[match.group(1)] = {"overall": {"count": int(count.group(1)) if count else 0,
                                                "state": state.group(1) if state else None}}
    return sections


def _dev_details(section: str, overall: dict[str, Any]) -> str:
    details = overall.get("details") if isinstance(overall.get("details"), dict) else {}
    if section == "pullrequest":
        parts = [f"{details.get(k)} {label}" for k, label in
                 (("openCount", "open"), ("mergedCount", "merged"), ("declinedCount", "declined")) if details.get(k)]
        if parts:
            return ", ".join(parts)
        return str(overall.get("state") or "").lower()
    if section == "build":
        return ", ".join(f"{overall.get(k)} {label}" for k, label in
                         (("successfulBuildCount", "successful"), ("failedBuildCount", "failed"),
                          ("unknownBuildCount", "unknown")) if overall.get(k))
    if section == "deployment-environment":
        names = [str(env.get("title") or env.get("name")) for env in overall.get("topEnvironments") or []
                 if isinstance(env, dict) and (env.get("title") or env.get("name"))]
        return ", ".join(names)
    return ""


def format_dev_summary(value: Any) -> str:
    """``1 branch, 3 commits, 1 pull request (merged) - Bitbucket Server``, or "" when there is nothing."""
    summary = _dev_summary_data(value)
    if summary is None:
        summary = _dev_counts_from_text(value) if isinstance(value, str) else {}
    parts: list[str] = []
    sources: list[str] = []
    for section, singular, plural in _DEV_SECTIONS:
        item = summary.get(section) if isinstance(summary.get(section), dict) else {}
        overall = item.get("overall") if isinstance(item.get("overall"), dict) else {}
        try:
            count = int(overall.get("count") or 0)
        except (TypeError, ValueError):
            count = 0
        if count <= 0:
            continue
        text = f"{count} {singular if count == 1 else plural}"
        details = _dev_details(section, overall)
        parts.append(f"{text} ({details})" if details else text)
        for instance in (item.get("byInstanceType") or {}).values():
            name = instance.get("name") if isinstance(instance, dict) else None
            if name and name not in sources:
                sources.append(str(name))
    if not parts:
        return ""
    return ", ".join(parts) + (f" - {', '.join(sources)}" if sources else "")
