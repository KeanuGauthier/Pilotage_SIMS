def _with_slow_first_click(ui: Any, delay_ms: int, action: Callable[[], Any]) -> Any:
    """Delays the first click, like a slow machine on a heavy first page load."""

    original = ui.click
    state = {"delayed": False}

    def click(*args: Any, **kwargs: Any) -> Any:
        if not state["delayed"]:
            state["delayed"] = True
            ui.page.wait_for_timeout(delay_ms)
        return original(*args, **kwargs)

    ui.click = click
    try:
        return action()
    finally:
        del ui.click


class _Handler(BaseHTTPRequestHandler):
    """Serves the trap pages over real HTTP (redirects, downloads, errors)."""

    def log_message(self, *args: Any) -> None:  # keep the console quiet
        pass

    def _send(self, status: int, body: bytes, headers: Optional[dict] = None) -> None:
        self.send_response(status)
        for key, value in (headers or {}).items():
            self.send_header(key, value)
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self) -> None:  # noqa: N802 (http.server API)
        path = urlparse(self.path).path
        html = {"Content-Type": "text/html; charset=utf-8"}
        if path in PAGES:
            page = "<!doctype html><html><body>{}</body></html>".format(PAGES[path])
            self._send(200, page.encode("utf-8"), html)
        elif path == "/api/fail":
            self._send(500, b"boom", {"Content-Type": "text/plain"})
        elif path == "/img/ok.png":
            self._send(200, PNG_1PX, {"Content-Type": "image/png"})
        elif path == "/app":
            port = self.server.server_address[1]
            if "sso=ok" in (self.headers.get("Cookie") or ""):
                body = '<meta name="ajs-remote-user" content="jdoe"><div id="dashboard">Dashboard</div>'
                self._send(200, "<!doctype html><html><body>{}</body></html>".format(body).encode(), html)
            else:
                here = "http://127.0.0.1:{}{}".format(port, self.path)
                chooser = "http://localhost:{}/chooser?next={}".format(port, quote(here, safe=""))
                self._send(302, b"", {"Location": chooser})
        elif path == "/app-flaky":
            port = self.server.server_address[1]
            here = "http://127.0.0.1:{}/app".format(port)
            chooser = "http://localhost:{}/chooser-flaky?next={}".format(port, quote(here, safe=""))
            self._send(302, b"", {"Location": chooser})
        elif path == "/sso-callback":
            target = parse_qs(urlparse(self.path).query)["next"][0]
            self._send(302, b"", {"Location": target, "Set-Cookie": "sso=ok; Path=/"})
        elif path == "/app-stuck":
            self._send(302, b"", {"Location": "http://localhost:{}/idp-stuck".format(self.server.server_address[1])})
        elif path == "/sso-broken":
            self._send(302, b"", {"Location": "/login.jsp?os_destination=%2Fdashboard"})
        elif path == "/secure":
            self._send(302, b"", {"Location": "/login.jsp?os_destination=%2Fsecure"})
        elif path == "/file/report.pdf":
            self._send(200, b"%PDF-1.4\n%fake but well-formed header\n", {
                "Content-Type": "application/pdf",
                "Content-Disposition": "attachment; filename=report.pdf"})
        elif path == "/file/fake.pdf":
            self._send(200, b"<!DOCTYPE html><html><body>Internal error</body></html>", {
                "Content-Type": "application/pdf",
                "Content-Disposition": "attachment; filename=fake.pdf"})
        else:
            self._send(404, b"<h1>Not found</h1>", html)


def start_server() -> Tuple[ThreadingHTTPServer, str]:
    server = ThreadingHTTPServer(("127.0.0.1", 0), _Handler)
    threading.Thread(target=server.serve_forever, daemon=True).start()
    return server, "http://127.0.0.1:{}".format(server.server_address[1])


def assert_true(value: Any) -> None:
    if not value:
        raise UiCheckError("expected a truthy value, got {!r}".format(value))


class SelfTest:
    def __init__(self, ui: Actions) -> None:
        self.ui = ui
        self.results: List[Tuple[str, bool, str]] = []

    def open(self, path: str) -> None:
        self.ui.goto(ORIGIN + path)

    def must_pass(self, name: str, check: Callable[[], Any]) -> None:
        started = time.monotonic()
        try:
            check()
        except Exception as error:
            detail = "{}: {}".format(type(error).__name__, str(error).splitlines()[0][:160])
            self.results.append((name, False, "expected PASS, got " + detail))
            traceback.print_exc()
            return
        self.results.append((name, True, "passed ({:.1f}s)".format(time.monotonic() - started)))

    def must_fail(self, name: str, check: Callable[[], Any], contains: Optional[str] = None) -> None:
        started = time.monotonic()
        try:
            check()
        except UiCheckError as error:
            message = str(error)
            if contains and contains.lower() not in message.lower():
                self.results.append((name, False, "failed but message lacks {!r}: {}".format(
                    contains, message.splitlines()[0][:160])))
                return
            first = message.splitlines()[0][:110]
            self.results.append((name, True, "failed as expected ({:.1f}s): {}".format(
                time.monotonic() - started, first)))
            return
        except Exception as error:
            self.results.append((name, False, "expected UiCheckError, got {}: {}".format(
                type(error).__name__, str(error).splitlines()[0][:160])))
            return
        self.results.append((name, False, "expected FAIL, but the check passed (false positive!)"))


def run(headed: bool) -> int:
    global ORIGIN
    server, ORIGIN = start_server()
    channel = os.getenv("BROWSER_CHANNEL", "msedge").strip() or None
    download_dir = Path(tempfile.mkdtemp(prefix="nrt-selftest-"))
    with sync_playwright() as playwright:
        browser = playwright.chromium.launch(channel=channel, headless=not headed)
        context = browser.new_context(viewport={"width": 1280, "height": 800}, accept_downloads=True)
        page = context.new_page()
        ui = Actions(
            page,
            timeout_ms=4_000,
            navigation_timeout_ms=10_000,
            ready_timeout_ms=3_000,
            settle_ms=200,
            click_retry_after_ms=900,
            artifacts_dir=download_dir,
        )
        t = SelfTest(ui)

        # --- waiting & visibility -------------------------------------------------
        t.open("/delayed")
        t.must_pass("element appearing after 1.2s is found", lambda: ui.expect_visible("#late"))

        t.open("/opacity")
        t.must_pass("normal button is visible", lambda: ui.expect_visible("#ok"))
        t.must_fail("opacity:0 button is NOT visible", lambda: ui.expect_visible("#ghost", timeout_ms=1_000),
                    "transparent")
        t.must_fail("element in opacity:0 parent is NOT visible",
                    lambda: ui.expect_visible("#ghost2", timeout_ms=1_000), "transparent")
        t.must_pass("(info) plain Playwright wrongly says opacity:0 is visible",
                    lambda: expect(page.locator("#ghost")).to_be_visible(timeout=1_000))

        t.open("/covered")
        t.must_fail("element under an overlay is NOT visible",
                    lambda: ui.expect_visible("#target", timeout_ms=1_000), "covered by div#overlay")

        t.open("/clipped")
        t.must_fail("element clipped by overflow:hidden is NOT visible",
                    lambda: ui.expect_visible("#c", timeout_ms=1_000))

        # --- iframes ---------------------------------------------------------------
        deep = Target("#deep", frame=["#outer", "iframe.inner"])
        t.open("/frames")
        t.must_pass("click in nested iframes with post-condition",
                    lambda: ui.click(deep, expect=Target("#deep", frame=["#outer", "iframe.inner"],
                                                         has_text="Clicked")))
        t.must_pass("ANY_FRAME finds the element in nested iframes",
                    lambda: ui.expect_visible(Target("#deep", frame=ANY_FRAME)))
        t.must_fail("missing frame= gives a hint where the element is",
                    lambda: ui.expect_visible("#deep", timeout_ms=800), "use frame=")

        t.open("/zero-frame")
        t.must_fail("content of a 0px-high iframe is NOT visible",
                    lambda: ui.expect_visible(Target("#deep", frame="#z"), timeout_ms=1_000))

        # --- clicks ----------------------------------------------------------------
        t.open("/hidden-dup")
        t.must_pass("hidden duplicate is ignored (no strict-mode error)",
                    lambda: ui.click(".save", expect=Target("#res", has_text="saved")))

        t.open("/strict")
        t.must_fail("two visible matches are refused", lambda: ui.click(".btn", timeout_ms=1_000), "SEVERAL")

        t.open("/alternatives")
        t.must_pass("alternatives: the matching one is used",
                    lambda: ui.click(["#old", "#new"], expect=Target("#new", has_text="done")))

        def late_handler_with_retry() -> None:
            ui.click_retry_after_ms = 1_500
            try:
                ui.begin_step()
                ui.click("#open", expect="#menu")
                assert any("clicking again" in warning for warning in ui.warnings), ui.warnings
            finally:
                ui.click_retry_after_ms = 900

        t.open("/late-handler")
        t.must_pass("click before JS is bound is retried (warning recorded)", late_handler_with_retry)
        t.open("/late-handler")
        t.must_fail("with retries=0 the same click fails",
                    lambda: ui.click("#open", expect="#menu", retries=0, timeout_ms=1_500), "did not happen")

        # --- typing ----------------------------------------------------------------
        t.open("/react-input")
        t.must_pass("input rejecting the first set is retyped and verified",
                    lambda: (ui.fill("#r", "hello"), ui.expect_value("#r", "hello")))
        t.open("/uppercase")
        t.must_fail("value rewritten by the app is detected", lambda: ui.fill("#u", "abc"), "contains 'ABC'")

        t.must_pass("expected_value= accepts a legit reformat", lambda: ui.fill("#u", "abc", expected_value="ABC"))
