"""Start-up helpers shared by the command-line scripts.

This module uses the Python standard library only: it runs *before* the
third-party libraries are imported, possibly with a Python that does not have them.

Why it exists: on the company workstations ``.bat`` files are blocked, so the
scripts are started directly with ``py <script>.py``. That command uses the
system-wide Python. :func:`use_project_venv` makes the script re-start itself
with the private Python environment (``.venv``) created by ``install.py``, so
that users never have to "activate" anything.
"""

from __future__ import annotations

import os
import subprocess
import sys
from pathlib import Path

# Set for the re-started process, so it never tries to re-start itself again.
_SWITCHED_ENV = "JIRA_ARCHIVER_IN_VENV"


def venv_python(project_root: Path) -> Path:
    """Path of the Python program inside the project's ``.venv`` folder."""
    if os.name == "nt":
        return project_root / ".venv" / "Scripts" / "python.exe"
    return project_root / ".venv" / "bin" / "python"


def _running_in(venv_dir: Path) -> bool:
    try:
        return Path(sys.prefix).resolve() == venv_dir.resolve()
    except OSError:
        return False


def started_by_double_click() -> bool:
    """Best-effort detection of a script opened by double-click in Windows Explorer.

    In that case Windows opens a new console window that closes as soon as the
    script ends, so the messages could not be read. A console opened by
    double-click only holds the Python processes (at most 2: the ``py`` launcher
    and Python), whereas a terminal also holds ``cmd.exe`` or ``powershell.exe``.
    """
    if os.name != "nt" or os.environ.get(_SWITCHED_ENV) or not sys.stdin or not sys.stdin.isatty():
        return False
    try:
        import ctypes

        process_list = (ctypes.c_uint * 16)()
        count = ctypes.windll.kernel32.GetConsoleProcessList(process_list, 16)
    except Exception:  # noqa: BLE001 - detection is optional, never fail because of it
        return False
    return 0 < count <= 2


def pause_if_double_clicked(double_clicked: bool) -> None:
    """Keep the console window open so the user can read the result."""
    if double_clicked:
        try:
            input("\nPress Enter to close this window...")
        except (EOFError, KeyboardInterrupt):
            pass


def use_project_venv(project_root: Path) -> None:
    """Re-run the current script with ``.venv`` Python if it exists and is not the one running.

    Returns normally (and the script continues with the current Python) when there
    is no ``.venv`` folder, when it is already the running Python, or when the
    ``.venv`` Python cannot be started (e.g. blocked by a company policy). Otherwise
    it never returns: it exits with the exit code of the re-started script.
    """
    venv_dir = project_root / ".venv"
    python = venv_python(project_root)
    if os.environ.get(_SWITCHED_ENV) or not python.is_file() or _running_in(venv_dir):
        return

    double_clicked = started_by_double_click()
    script = str(Path(sys.argv[0]).resolve())
    env = dict(os.environ, **{_SWITCHED_ENV: "1"})
    sys.stdout.flush()
    try:
        # Fixed program (our own .venv Python) and our own script: no shell, no user-built command.
        process = subprocess.Popen([str(python), script, *sys.argv[1:]], env=env)  # noqa: S603
    except OSError as exc:
        print(f"[WARN] Could not start the Python of the .venv folder ({exc}).")
        print("       Continuing with the current Python. If this keeps happening, see")
        print('       "Installation without .venv" in README.md.')
        return

    while True:
        try:
            exit_code = process.wait()
            break
        except KeyboardInterrupt:
            # Ctrl+C also reaches the re-started script, which stops cleanly by itself.
            continue
    pause_if_double_clicked(double_clicked)
    sys.exit(exit_code)
